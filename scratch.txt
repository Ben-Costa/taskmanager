 6. Example model using better‑sqlite3
This is the clean, low‑debt pattern you prefer:
// models/notes.js
const db = require('../db');

module.exports = {
  getAll() {
    return db.prepare('SELECT * FROM notes').all();
  },

  getById(id) {
    return db.prepare('SELECT * FROM notes WHERE id = ?').get(id);
  },

  create({ title, body }) {
    const stmt = db.prepare(`
      INSERT INTO notes (title, body)
      VALUES (?, ?)
    `);
    const result = stmt.run(title, body);
    return { id: result.lastInsertRowid, title, body };
  }
};
